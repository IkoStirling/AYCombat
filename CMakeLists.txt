cmake_minimum_required(VERSION 3.25)
project(AYCombat VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(PROJECT_IS_TOP_LEVEL AND MSVC)
    add_compile_options(/utf-8)
    if(CMAKE_GENERATOR MATCHES "Ninja")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/combat-prefix.h" "#pragma once\n")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/combat-prefix.cpp" "#include \"combat-prefix.h\"\n")
        execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /nologo /utf-8 /showIncludes /Zs "${CMAKE_CURRENT_BINARY_DIR}/combat-prefix.cpp"
            OUTPUT_VARIABLE _combat_includes ERROR_VARIABLE _combat_errors ENCODING UTF-8)
        string(APPEND _combat_includes "${_combat_errors}")
        if(_combat_includes MATCHES "注意: 包含文件:  ")
            set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
        elseif(_combat_includes MATCHES "Note: including file:  ")
            set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:  ")
        else()
            message(FATAL_ERROR "Cannot detect MSVC include prefix for Ninja dependency tracking")
        endif()
        set(CMAKE_C_CL_SHOWINCLUDES_PREFIX "${CMAKE_CL_SHOWINCLUDES_PREFIX}")
        set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${CMAKE_CL_SHOWINCLUDES_PREFIX}")
        add_compile_definitions(AY_COMBAT_NINJA_DEPS_PREFIX=1)
    endif()
endif()
add_library(AYCombatCore STATIC src/Combat.cpp src/HitLedger.cpp src/Interaction.cpp src/Rules.cpp src/InitiativeClock.cpp)
add_library(AYCombat::Core ALIAS AYCombatCore)
target_compile_features(AYCombatCore PUBLIC cxx_std_23)
target_include_directories(AYCombatCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
add_executable(AYCombat_InitiativeExample examples/Initiative.cpp)
target_link_libraries(AYCombat_InitiativeExample PRIVATE AYCombatCore)
if(MSVC)
    target_compile_options(AYCombatCore PRIVATE /W4 /utf-8 /Zc:__cplusplus)
endif()
option(AY_COMBAT_BUILD_STATS "Build the synchronous single-StatSet settlement adapter" ON)
option(AY_COMBAT_BUILD_ENTITY "Build World/StatsComponent combat integration (engine build)" OFF)
if(PROJECT_IS_TOP_LEVEL AND AY_COMBAT_BUILD_ENTITY)
    message(FATAL_ERROR "AY_COMBAT_BUILD_ENTITY requires the root engine build")
endif()
if(BUILD_TESTING AND NOT TARGET AYTest)

# Called by the engine after AYEntity and AYStats::Entity have been assembled.
function(ay_combat_add_entity_integration)
    if(NOT AY_COMBAT_BUILD_ENTITY)
        return()
    endif()
    if(NOT TARGET AYCombatStatsIntegration OR NOT TARGET AYEntityStatsIntegration)
        message(FATAL_ERROR "AY_COMBAT_BUILD_ENTITY requires AYCombat::Stats and AYStats::Entity")
    endif()
    add_library(AYEntityCombatIntegration STATIC "${AY_COMBAT_SOURCE_DIR}/src/EntityIntegration.cpp")
    add_library(AYCombat::Entity ALIAS AYEntityCombatIntegration)
    target_link_libraries(AYEntityCombatIntegration PUBLIC AYCombatStatsIntegration AYEntityStatsIntegration)
    if(MSVC)
        target_compile_options(AYEntityCombatIntegration PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
    if(BUILD_TESTING)
        add_executable(AYCombat_EntityTests "${AY_COMBAT_SOURCE_DIR}/unittest/main.cpp" "${AY_COMBAT_SOURCE_DIR}/unittest/Test_EntityIntegration.cpp" "${AY_COMBAT_SOURCE_DIR}/unittest/Test_Gameplay.cpp")
        target_link_libraries(AYCombat_EntityTests PRIVATE AYEntityCombatIntegration AYTest)
        target_include_directories(AYCombat_EntityTests PRIVATE "${AY_COMBAT_SOURCE_DIR}/../AY2D/include")
        if(MSVC)
            target_compile_options(AYCombat_EntityTests PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
        endif()
        ay_add_test(AYCombat_EntityTests CASE_REPORT LABELS "combat;combat-integration;combat-full;integration" TIMEOUT 60 RUN_SERIAL)
    endif()
    add_executable(AYCombat_GameplayExample "${AY_COMBAT_SOURCE_DIR}/examples/Gameplay.cpp")
    target_link_libraries(AYCombat_GameplayExample PRIVATE AYEntityCombatIntegration)
    if(MSVC)
        target_compile_options(AYCombat_GameplayExample PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
endfunction()
function(ay_combat_add_physics2d_integration)
    if(NOT AY_COMBAT_BUILD_ENTITY)
        return()
    endif()
    if(NOT TARGET AYEntityCombatIntegration OR NOT TARGET AYPhysics)
        message(FATAL_ERROR "AYCombat 2D area queries require Entity integration and AYPhysics")
    endif()
    add_library(AYCombatPhysics2D STATIC "${AY_COMBAT_SOURCE_DIR}/src/Physics2DArea.cpp")
    add_library(AYCombat::Physics2D ALIAS AYCombatPhysics2D)

        AYEntityPhysicsIntegration AYPhysics)
    if(MSVC)
        target_compile_options(AYCombatPhysics2D PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
endfunction()

function(ay_combat_write_test_features)
    set(_combat_scene OFF)
    set(_combat_host OFF)
    if(TARGET AYCombatSceneIntegration)
        set(_combat_scene ON)
    endif()
    if(TARGET AYCombat_HostTests)
        set(_combat_host ON)
    endif()
    file(WRITE "${AY_COMBAT_BINARY_DIR}/unittest/test-features.json" "{\"stats\":\"${AY_COMBAT_BUILD_STATS}\",\"entity\":\"${AY_COMBAT_BUILD_ENTITY}\",\"scene\":\"${_combat_scene}\",\"host\":\"${_combat_host}\"}\n")
endfunction()
function(ay_combat_add_scene_integration)
    if(TARGET AYEntityCombatIntegration AND TARGET AYScene)
        add_library(AYCombatSceneIntegration STATIC "${AY_COMBAT_SOURCE_DIR}/src/SceneIntegration.cpp")
        add_library(AYCombat::Scene ALIAS AYCombatSceneIntegration)
        target_link_libraries(AYCombatSceneIntegration PUBLIC AYEntityCombatIntegration AYScene AYEventSystem)
        if(TARGET AYCombat_EntityTests)
            target_sources(AYCombat_EntityTests PRIVATE "${AY_COMBAT_SOURCE_DIR}/unittest/Test_Scene.cpp")
            target_link_libraries(AYCombat_EntityTests PRIVATE AYCombatSceneIntegration)
        endif()
    endif()
    ay_combat_write_test_features()
endfunction()
function(ay_combat_add_host_integration)
    if(TARGET AYCombatSceneIntegration AND TARGET AYApplication)
        add_library(AYCombatHostIntegration STATIC
            "${AY_COMBAT_SOURCE_DIR}/src/HostIntegration.cpp"
            "${AY_COMBAT_SOURCE_DIR}/src/Action2DWorldCommit.cpp"
            "${AY_COMBAT_SOURCE_DIR}/src/Action2DWorldSnapshot.cpp")
        add_library(AYCombat::Host ALIAS AYCombatHostIntegration)
        target_link_libraries(AYCombatHostIntegration PUBLIC AYCombatSceneIntegration
            AYApplication AYApplicationSaveGame AY2D AYStatsAssets)
        target_link_libraries(AYCombatHostIntegration PRIVATE nlohmann_json::nlohmann_json)
        if(TARGET AYCombat_EntityTests)
            add_executable(AYCombat_HostTests "${AY_COMBAT_SOURCE_DIR}/unittest/main.cpp" "${AY_COMBAT_SOURCE_DIR}/unittest/Test_Host.cpp" "${AY_COMBAT_SOURCE_DIR}/unittest/Test_Action2DWorldCommit.cpp")
            target_link_libraries(AYCombat_HostTests PRIVATE AYCombatHostIntegration AYTest)
            if(MSVC)
                target_compile_options(AYCombat_HostTests PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
            endif()
            ay_add_test(AYCombat_HostTests CASE_REPORT LABELS "combat;combat-integration;combat-full;integration" TIMEOUT 60 RUN_SERIAL)
        endif()
    endif()
    ay_combat_write_test_features()
endfunction()
